Цикл - это многократное повторение одного и того же действия до тех пор, пока выполняется какое-либо условие

# while

```c
while (...) {
	...
}
```

Пока выполняется условие в круглых скобках - код будет выполняться повторно

```c
#include <stdio.h>

int main(void) {
	int a = 5;
	while (a > 0) {
		printf("Осталось %d сек\n", a);
		a--;
	}
	return 0;
}
```

```c
#include <stdio.h>

int main(void) {
	int a = 3294122;
	
	while (a != 0) {
		printf("%d ", a % 10);
		a /= 10;
	}
}
```

```c
#include <stdio.h>

int main(void) {
    int age = -1; 
    while (age <= 0 || age > 120) {
        printf("Пожалуйста, введите ваш реальный возраст: ");
        scanf("%d", &age);

        if (age <= 0 || age > 120) {
            printf("Ошибка! Возраст должен быть от 1 до 120 лет.\n\n");
        }
    }

    printf("Спасибо! Ваш возраст принят: %d\n", age);
    return 0;
}
```

```c
#include <stdio.h>

int main(void) {
	int i = 0;
	while (i < 100) {
		printf("%d ", i);
		i += 2;
	}
	return 0;
}
```

# for

Когда условие состоит лишь в том, чтобы какое-то число добралось до определенного значения - нам облегчает жизнь `for`

```c
for (<1>; <2>; <3>) {
	...
}
```
1 - Инициализационное выражение (то, что выполнится до первого шага цикла)
2 - Условие (то, до какого момента цикл будет продолжаться)
3 - Шаг (действие, которое будет выполняться после каждого шага цикла)

Все эти три действия можно найти в предыдущем примере:
```c
	int i = 0; // 1
	while (i < 100) { // 2
		printf("%d ", i);
		i += 2; // 3
	}
```
И они объединяются в одну удобную строку

```c
for (int i = 0; i < 100; i += 2) {
	printf("%d ", i);
}
```

```c
#include <stdio.h>

int main(void) {
	for (int i = 0; i < 5; i++) {
		printf("Hello world");
	}
}
```

```c
#include <stdio.h>

int main(void) {
	for (int i = 10; i >= 0; i--) {
		printf("ostalos %d sec\n", i);
	}
	return 0;
}
```

```c
#include <stdio.h>

int main(void) {
	for(int i = 0, j = 10; i < 5 && j > 2; i++, j-=3) {
		printf("i=%d, j=%d\n", i, j);
	}
	return 0;
}
```

```c
#include <stdio.h>

int main(void) {
	for(int i = 0; i < 5; i++) {
		for(int j = 0; j < 10; j++) {
			printf("0");
		}
		printf("\n");
	}
	return 0;
}
```

```c
#include <stdio.h>

int main(void) {
	for(int i = 0; i < 5; i++) {
		for(int j = 0; j < 5; j++) {
			if (i >= j) printf("-");
			else printf("0");
		}
		printf("\n");
	}
	return 0;
}
```

# do while

Аналогичен while, но условие проверяется **ПОСЛЕ** действия
*то есть одна итерация выполнится гарантированно*

```c
#include <stdio.h>

int main() {
    int choice;

    do {
        printf("\n--- МЕНЮ БАНКОМАТА ---\n");
        printf("1. Проверить баланс\n");
        printf("2. Снять наличные\n");
        printf("3. Выйти\n");
        printf("Выберите действие: ");
        
        scanf("%d", &choice);

        if (choice == 1) {
            printf("Ваш баланс: 5000 руб.\n");
        } else if (choice == 2) {
            printf("Деньги выданы.\n");
        } else if (choice != 3) {
            printf("Неверный ввод!\n");
        }

    } while (choice != 3); // Проверка в конце. Если choice == 3, цикл завершается.

    printf("Спасибо, что воспользовались нашим банкоматом!\n");
    return 0;
}
```
