Позволяет делать нелинейные программы

Может быть краткой формы, без else:
```c
if (...) {
	...
}
```

Если в результате необходимо совершить только одно действие - фигурные скобки можно опустить
Например:
```c
int main(void) {
	int a = -5;
	if (a < 0)
		a = -a;
	return 0;
}
```

```c
#include <stdio.h>
int main(void) {
	int a = 0;
	int b;
	
	if (a > 0) b = 1;
	else
		if (a < 0) b = -1;
		else b = 0;
	
	printf("%d", b);
	return 0;
}
```
В этом примере внимательный зритель заметит, что мы "изобрели" последующую проверку условия, которая запустится только при невыполнении всех верхних блоков - `else if`


```c
int main(void) {
	int a = -5;
	int b;
	
	if (a > 0) {
		b = 1;
	}
	else if (a < 0) {
		b = -1;
	}	
	else {
		b = 0;
	}
			
	return 0;
}
```

*Зачем нам else if если можно просто написать много if?*
Пример:
```c
#include <stdio.h>

int main(void) {
	int a = 8;
	if (a >= 8) printf("Otl\n");
	if (a >= 6) printf("Hor\n");
	if (a >= 4) printf("Norm\n");
	else printf("IUP\n");
	
	return 0;
}
```

```c
#include <stdio.h>

int main(void) {
	int a = 8;
	if (a >= 8) printf("Otl\n");
	else if (a >= 6) printf("Hor\n");
	else if (a >= 4) printf("Norm\n");
	else printf("IUP\n");
	
	return 0;
}
```

И еще пример:

```c
int main(void) {
	int hp = 100;

	// Игрок упал и получил 60 урона
	hp -= 60; // Сейчас hp = 40
	
	if (hp > 50) {
	    cout << "Игрок ранен, но держится.\n";
	}
	if (hp <= 50 && hp > 0) {
	    cout << "Игрок при смерти! Срочно нужна аптечка!\n"; // Сработает!
	    hp -= 40; // Здоровье падает от кровотечения. Теперь hp = 0!
	}
	if (hp <= 0) {
	    cout << "Игрок погиб.\n"; // ТОЖЕ СРАБОТАЕТ в этот же миг!
	}
}
```
